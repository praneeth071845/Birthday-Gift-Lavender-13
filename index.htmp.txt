<script>
function celebrate() {
    const duration = 5 * 1000;
    const end = Date.now() + duration;

    (function frame() {
        confetti({
            particleCount: 6,
            spread: 80,
            startVelocity: 35,
            origin: {
                x: Math.random(),
                y: Math.random() * 0.5
            }
        });

        if (Date.now() < end) {
            requestAnimationFrame(frame);
        }
    })();

    // Big confetti burst
    confetti({
        particleCount: 150,
        spread: 120,
        origin: { y: 0.6 }
    });
}

// Automatic confetti when website opens
window.addEventListener("load", () => {
    setTimeout(() => {
        confetti({
            particleCount: 180,
            spread: 100,
            origin: { y: 0.6 }
        });
    }, 700);
});
